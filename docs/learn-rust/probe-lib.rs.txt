use chacha20poly1305::{aead::{Aead, KeyInit, Payload}, ChaCha20Poly1305, Nonce};
use spake2::{Ed25519Group, Identity, Password, Spake2};

#[cfg(target_arch = "wasm32")]
#[cfg(target_arch = "wasm32")]
#[link(wasm_import_module = "env")]
unsafe extern "C" { fn ls_random(ptr: *mut u8, len: usize); }

#[cfg(target_arch = "wasm32")]
fn js_random(buf: &mut [u8]) -> Result<(), getrandom::Error> {
    unsafe { ls_random(buf.as_mut_ptr(), buf.len()) };
    Ok(())
}
#[cfg(target_arch = "wasm32")]
getrandom::register_custom_getrandom!(js_random);

#[unsafe(no_mangle)]
pub extern "C" fn probe_seal(len: usize) -> usize {
    let key = [7u8; 32];
    let cipher = ChaCha20Poly1305::new(&key.into());
    let data = vec![1u8; len];
    let out = cipher.encrypt(Nonce::from_slice(&[0u8; 12]), Payload { msg: &data, aad: b"x" }).unwrap();
    out.len()
}

#[unsafe(no_mangle)]
pub extern "C" fn probe_pake() -> usize {
    let (a, ma) = Spake2::<Ed25519Group>::start_symmetric_with_rng(&Password::new(b"123456"), &Identity::new(b"id"), rand_core::OsRng);
    let (b, mb) = Spake2::<Ed25519Group>::start_symmetric_with_rng(&Password::new(b"123456"), &Identity::new(b"id"), rand_core::OsRng);
    let ka = a.finish(&mb).unwrap();
    let kb = b.finish(&ma).unwrap();
    (ka == kb) as usize
}
